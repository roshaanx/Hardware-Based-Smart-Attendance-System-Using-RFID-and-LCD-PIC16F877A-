#include <xc.h>
#define _XTAL_FREQ 8000000

// CONFIG
#pragma config FOSC = HS
#pragma config WDTE = OFF
#pragma config PWRTE = ON
#pragma config BOREN = ON
#pragma config LVP = OFF
#pragma config CPD = OFF
#pragma config WRT = OFF
#pragma config CP = OFF

// ---------- LCD Pins ----------
#define RS RB0
#define EN RB1
#define D4 RB2
#define D5 RB3
#define D6 RB4
#define D7 RB5

// ---------- Function Prototypes ----------
void LCD_Init();
void LCD_Command(unsigned char);
void LCD_Char(unsigned char);
void LCD_String(const char*);
void LCD_Clear();
char UART_Read();
void UART_Init();
void LCD_Int(int num);

// ---------- Attendance storage in GPR ----------
unsigned char *attendance = (unsigned char*)0x20; // File registers
unsigned char index = 0;

// ---------- RFID Variables ----------
int receivedNum = 0;       // Card number
const int VALID_ID = 6162; // Valid card ID

// ---------- UART Init ----------
void UART_Init(void){
    TRISC7 = 1; // RX input
    TRISC6 = 0; // TX output
    SPBRG = 51; // 9600 baud @ 8MHz
    BRGH = 1;
    SYNC = 0;
    SPEN = 1;
    CREN = 1;
    TXEN = 1;
}

// ---------- UART Read ----------
char UART_Read(){
    while(!RCIF);
    return RCREG;
}

// ---------- LCD Functions ----------
void LCD_Pulse(){
    EN = 1;
    __delay_ms(1);
    EN = 0;
}

void LCD_Command(unsigned char cmd){
    RS = 0;
    D4 = (cmd>>4)&1;
    D5 = (cmd>>5)&1;
    D6 = (cmd>>6)&1;
    D7 = (cmd>>7)&1;
    LCD_Pulse();

    D4 = cmd&1;
    D5 = (cmd>>1)&1;
    D6 = (cmd>>2)&1;
    D7 = (cmd>>3)&1;
    LCD_Pulse();
}

void LCD_Char(unsigned char data){
    RS = 1;
    D4 = (data>>4)&1;
    D5 = (data>>5)&1;
    D6 = (data>>6)&1;
    D7 = (data>>7)&1;
    LCD_Pulse();

    D4 = data&1;
    D5 = (data>>1)&1;
    D6 = (data>>2)&1;
    D7 = (data>>3)&1;
    LCD_Pulse();
}

void LCD_String(const char *str){
    while(*str)
        LCD_Char(*str++);
}

void LCD_Int(int num){
    char buf[6];
    int i=0,j;
    int temp = num;

    if(num == 0){
        LCD_Char('0');
        return;
    }

    while(temp > 0){
        buf[i++] = (temp%10) + '0';
        temp /= 10;
    }
    buf[i] = '\0';

    for(j=i-1; j>=0; j--){
        LCD_Char(buf[j]);
    }
}

void LCD_Init(){
    TRISB = 0x00;
    __delay_ms(20);
    LCD_Command(0x02);
    LCD_Command(0x28);
    LCD_Command(0x0C);
    LCD_Command(0x06);
    LCD_Command(0x01);
}

void LCD_Clear(){
    LCD_Command(0x01);
    __delay_ms(2);
}

// ---------- MAIN FUNCTION ----------
void main(void){
    char i,c;

    // PIC pin setup
    TRISC0 = 0; // Buzzer output
    TRISC1 = 0; // LED output
    RC0 = 0;
    RC1 = 0;

    LCD_Init();
    UART_Init();

    LCD_String("RFID Attendance");
    __delay_ms(1000);
    LCD_Clear();

    while(1){
        receivedNum = 0;

        // Read 4 characters from Virtual Terminal and convert to int
        for(i=0;i<4;i++){
            c = UART_Read();
            receivedNum = receivedNum*10 + (c-'0');
        }

        // Display scanned card number
        LCD_Clear();
        LCD_String("Card:");
        LCD_Command(0xC0);
        LCD_Int(receivedNum);

        // Check if valid
        if(receivedNum == VALID_ID){
            LCD_Clear();
            LCD_String("Attendance");
            LCD_Command(0xC0);
            LCD_String("Marked!");
            RC1 = 1; // LED ON
            __delay_ms(300);
            RC1 = 0;

            // Store in RAM GPR
            attendance[index] = 1;
            index++;

            // Optional: display attendance summary
            LCD_Clear();
            LCD_String("Attendance:");
            for(i=0;i<index;i++){
                LCD_Command(0xC0 + i); // Adjust for second line
                LCD_Char('P'); // P for present
            }

        } else {
            LCD_Clear();
            LCD_String("Invalid Card!");
            RC0 = 1; // Buzzer ON
            __delay_ms(300);
            RC0 = 0;
        }

        __delay_ms(500);
    }
}
